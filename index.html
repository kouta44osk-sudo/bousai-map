// 洪水浸水想定タイル（想定最大規模 L2） L.tileLayer('https://disaportaldata.gsi.go.jp/raster/01_flood_l2_shinsuishin/{z}/{x}/{y}.png', { attribution: '国土交通省 洪水浸水想定区域図', opacity: 0.6, transparent: true }).addTo(map); // 4. Geolocation APIで現在地を取得 if (navigator.geolocation) { navigator.geolocation.watchPosition( (position) => { const lat = position.coords.latitude; const lng = position.coords.longitude; const accuracy = position.coords.accuracy; // 精度（メートル） const latLng = [lat, lng]; // 既存のマーカー・円があれば削除 if (locationMarker) { map.removeLayer(locationMarker); } if (accuracyCircle) { map.removeLayer(accuracyCircle); } // 精度を表す薄い青色の円（外側の丸） accuracyCircle = L.circle(latLng, { radius: accuracy, color: '#136fad', fillColor: '#136fad', fillOpacity: 0.15, weight: 1 }).addTo(map); // 現在地を示す濃い青色の丸（中心の青丸） locationMarker = L.circleMarker(latLng, { radius: 8, color: '#ffffff', fillColor: '#2a81cb', fillOpacity: 1.0, weight: 2 }).addTo(map).bindPopup("現在地はこちらです").openPopup(); // 初回のみ地図の中心を現在地に移動 map.setView(latLng, 16); }, (error) => { alert("位置情報の取得に失敗しました: " + error.message); }, { enableHighAccuracy: true, maximumAge: 0, timeout: 10000 } ); } else { alert("お使いのブラウザは位置情報取得に対応していません。"); }
